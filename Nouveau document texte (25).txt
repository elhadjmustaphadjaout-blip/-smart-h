#include <WiFi.h>
#include <Firebase_ESP_Client.h>

#include <DHT.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// ================= WIFI =================
#define WIFI_SSID "YOUR_WIFI"
#define WIFI_PASSWORD "YOUR_PASS"

// ================= FIREBASE =================
#define API_KEY "YOUR_API_KEY"
#define DATABASE_URL "https://smartgreenhouse-a054e-default-rtdb.firebaseio.com/"

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// ================= SENSORS =================
#define DHTPIN 4
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

#define ONE_WIRE_BUS 5
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature ds18b20(&oneWire);

#define PH_PIN 34
#define TDS_PIN 35

// ================= TIMERS =================
unsigned long lastSend = 0;

// ================= VALUES =================
float h, t, dsTemp, ph, tds;

void setup() {
  Serial.begin(115200);
  delay(2000);

  // WiFi
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nWiFi Connected");

  // Firebase config
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  // Sensors
  dht.begin();
  ds18b20.begin();

  Serial.println("SYSTEM READY");
}

void readSensors() {

  h = dht.readHumidity();
  t = dht.readTemperature();

  ds18b20.requestTemperatures();
  dsTemp = ds18b20.getTempCByIndex(0);

  ph = map(analogRead(PH_PIN), 0, 4095, 0, 14);
  tds = map(analogRead(TDS_PIN), 0, 4095, 0, 3000);
}

void sendToFirebase() {

  if (Firebase.ready()) {

    Firebase.RTDB.setFloat(&fbdo, "/sensors/temperature", t);
    Firebase.RTDB.setFloat(&fbdo, "/sensors/humidity", h);
    Firebase.RTDB.setFloat(&fbdo, "/sensors/ds18b20", dsTemp);
    Firebase.RTDB.setFloat(&fbdo, "/sensors/ph", ph);
    Firebase.RTDB.setFloat(&fbdo, "/sensors/tds", tds);
  }
}

void loop() {

  // قراءة حساسات كل 2 ثانية
  if (millis() % 2000 < 50) {
    readSensors();

    Serial.print("T:");
    Serial.print(t);
    Serial.print(" H:");
    Serial.print(h);
    Serial.print(" PH:");
    Serial.print(ph);
    Serial.print(" TDS:");
    Serial.println(tds);
  }

  // إرسال Firebase كل 7 ثواني (مهم جداً)
  if (millis() - lastSend > 7000) {
    sendToFirebase();
    lastSend = millis();
  }
}
