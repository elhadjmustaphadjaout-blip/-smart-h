@Composable
fun SmartGreenHouseScreen(modifier: Modifier = Modifier) {

    var temperature by remember { mutableStateOf("0") }
    var humidity by remember { mutableStateOf("0") }
    var ph by remember { mutableStateOf("0") }
    var tds by remember { mutableStateOf("0") }

    LaunchedEffect(Unit) {

        val database = FirebaseDatabase.getInstance()

        while (true) {

            database.getReference("sensors/temperature")
                .get()
                .addOnSuccessListener {
                    temperature = it.value.toString()
                }

            database.getReference("sensors/humidity")
                .get()
                .addOnSuccessListener {
                    humidity = it.value.toString()
                }

            database.getReference("sensors/ph")
                .get()
                .addOnSuccessListener {
                    ph = it.value.toString()
                }

            database.getReference("sensors/tds")
                .get()
                .addOnSuccessListener {
                    tds = it.value.toString()
                }

            delay(2000)
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(20.dp)
    ) {

        Text(
            text = "SMART GREEN HOUSE",
            style = MaterialTheme.typography.headlineMedium
        )

        Spacer(modifier = Modifier.height(20.dp))

        Text("Temperature : $temperature °C")
        Text("Humidity : $humidity %")
        Text("pH : $ph")
        Text("TDS : $tds ppm")
    }
}
