// =====================================================
// قراءة أوامر التحكم اليدوي من Serial
// =====================================================

void readSerialCommand()
{
  if (Serial.available())
  {
    command = Serial.readStringUntil('\n');
    command.trim();

    // تبديل الوضع
    if (command == "AUTO")
    {
      AUTO_MODE = true;
      Serial.println("MODE: AUTO");
    }

    else if (command == "MANUAL")
    {
      AUTO_MODE = false;
      Serial.println("MODE: MANUAL");
    }

    // التحكم اليدوي (يعمل فقط في MANUAL)
    if (!AUTO_MODE)
    {
      if (command == "FAN ON") digitalWrite(FAN_RELAY, HIGH);
      if (command == "FAN OFF") digitalWrite(FAN_RELAY, LOW);

      if (command == "PUMP ON") digitalWrite(PUMP_RELAY, HIGH);
      if (command == "PUMP OFF") digitalWrite(PUMP_RELAY, LOW);

      if (command == "WINDOW ON") digitalWrite(WINDOW_RELAY, HIGH);
      if (command == "WINDOW OFF") digitalWrite(WINDOW_RELAY, LOW);

      if (command == "BUZZER ON") digitalWrite(BUZZER_PIN, HIGH);
      if (command == "BUZZER OFF") digitalWrite(BUZZER_PIN, LOW);
    }
  }
}

void loop()                                                 // الحلقة الرئيسية
{
  readSerialCommand();   // قراءة الأوامر في كل دورة
  hum = dht.readHumidity();                                // قراءة الرطوبة

  ds18b20.requestTemperatures();                           // طلب حرارة
  temp = ds18b20.getTempCByIndex(0);                      // قراءة الحرارة

  phRaw = analogRead(PH_PIN);                             // قراءة pH
  tdsRaw = analogRead(TDS_PIN);                            // قراءة TDS

  float tdsValue = map(tdsRaw, 0, 4095, 0, 4000);         // تحويل TDS

  digitalWrite(FERTILIZER_PUMP, LOW);                     // إطفاء السماد
  digitalWrite(FRESH_WATER_PUMP, LOW);                    // إطفاء الماء
  digitalWrite(GREEN_LED, LOW);                           // إطفاء الأخضر
  digitalWrite(YELLOW_LED, LOW);                          // إطفاء الأصفر
  digitalWrite(RED_LED, LOW);                             // إطفاء الأحمر

  if (tdsValue < 400)                                     // نقص غذاء
  {
    digitalWrite(FERTILIZER_PUMP, HIGH);                 // تشغيل السماد
    digitalWrite(YELLOW_LED, HIGH);                      // تحذير
  }

  else if (tdsValue >= 800 && tdsValue <= 1500)          // طبيعي
  {
    digitalWrite(GREEN_LED, HIGH);                       // حالة جيدة
  }

  else if (tdsValue > 2000 && tdsValue <= 3000)          // عالي
  {
    digitalWrite(FRESH_WATER_PUMP, HIGH);                // تخفيف
    digitalWrite(RED_LED, HIGH);                         // خطر

 else if (tdsValue > 3000)

  AUTO_MODE = true;

  digitalWrite(FRESH_WATER_PUMP, HIGH);

  digitalWrite(PUMP_RELAY, LOW);

}
  float phVoltage = (phRaw * 3.3) / 4095.0;              // تحويل الجهد

// =====================================================

{
  AUTO_MODE = true;  // إجبار النظام على التلقائي

  Firebase.setString(fbdo, "/control/mode", "AUTO");

  digitalWrite(FRESH_WATER_PUMP, HIGH);
  digitalWrite(RED_LED, HIGH);
  digitalWrite(BUZZER_PIN, HIGH);

  digitalWrite(PUMP_RELAY, LOW);

  Serial.println("EMERGENCY MODE ACTIVATED");
}
  // =====================================================  
  // التحقق من سلامة القراءات
  // =====================================================  

  if (isnan(hum) || temp == DEVICE_DISCONNECTED_C)        // إذا فشل الحساس
  {
    lcd.setCursor(0, 0);                                 // السطر الأول
    lcd.print("SENSOR ERROR    ");                       // خطأ حساسات

    lcd.setCursor(0, 1);                                 // السطر الثاني
    lcd.print("CHECK SENSORS   ");                       // تحقق من الحساسات

    Serial.println("Sensor Error");                      // طباعة خطأ

    delay(2000);                                          // انتظار
    return;                                               // إعادة loop من البداية
  }

  // =====================================================  
  // التحكم في الرطوبة
  // =====================================================  

  digitalWrite(FAN_RELAY, LOW);                          // إطفاء المروحة
  digitalWrite(WINDOW_RELAY, LOW);                       // إطفاء النافذة
  digitalWrite(PUMP_RELAY, LOW);                         // إطفاء المضخة

  if (AUTO_MODE && hum < 60)                                          // رطوبة منخفضة
  {
    if (millis() - lastSprayTime >= sprayInterval)       // وقت الرش
    {
      lastSprayTime = millis();                          // تحديث الوقت

      digitalWrite(PUMP_RELAY, HIGH);                  // تشغيل مضخة
      delay(sprayDuration);                             // مدة الرش
      digitalWrite(PUMP_RELAY, LOW);                   // إيقاف

      delay(3000);                                      // انتظار

      digitalWrite(PUMP_RELAY, HIGH);                  // رش 2
      delay(sprayDuration);
      digitalWrite(PUMP_RELAY, LOW);

      delay(3000);

      digitalWrite(PUMP_RELAY, HIGH);                  // رش 3
      delay(sprayDuration);
      digitalWrite(PUMP_RELAY, LOW);
    }
  }

  else if (hum >= 60 && hum <= 80)                      // طبيعي
  {
    // لا شيء
  }

  else if (AUTO_MODE && hum > 80)                                    // رطوبة عالية
  {
    digitalWrite(FAN_RELAY, HIGH);                     // تشغيل مروحة
    digitalWrite(WINDOW_RELAY, HIGH);                  // فتح نافذة
  }

  // =====================================================  
  // تغيير صفحات الشاشة
  // =====================================================  

  if (millis() - lastScreenChange >= 3000UL)           // كل 3 ثواني
  {
    lastScreenChange = millis();                       // تحديث الوقت

    if (screenPage == false)                           // تبديل الصفحة
      screenPage = true;
    else
      screenPage = false;

    lcd.clear();                                       // مسح الشاشة
  }

  // =====================================================  
  // الصفحة 1 (حرارة + رطوبة)
  // =====================================================  

  if (screenPage == false)
  {
    lcd.setCursor(0, 0);                              // السطر الأول
    lcd.print("T:");                                 // حرارة
    lcd.print(temp, 1);                              // قيمة الحرارة
    lcd.print("C ");

    lcd.setCursor(9, 0);                             // رطوبة
    lcd.print("H:");
    lcd.print(hum, 0);
    lcd.print("%");

    lcd.setCursor(0, 1);                             // السطر الثاني

    // لا يوجد نص إضافي مفعل حالياً (معلق)
  }

  // =====================================================  
  // الصفحة 2 (pH + TDS)
  // =====================================================  

  else
  {
    lcd.setCursor(0, 0);                             // pH
    lcd.print("pH:");
    lcd.print(phValue, 1);
    lcd.print("   ");

    lcd.setCursor(0, 1);                             // TDS
    lcd.print("TDS:");
    lcd.print((int)tdsValue);
    lcd.print("ppm");
  }

  // =====================================================  
  // إرسال البيانات للسيريال
  // =====================================================  

  Serial.print("Temperature = ");                    // حرارة
  Serial.print(temp);

  Serial.print(" C  Humidity = ");                  // رطوبة
  Serial.print(hum);

  Serial.print(" %  pH = ");                        // pH
  Serial.print(phValue, 1);

  Serial.print("  TDS = ");                         // TDS
  Serial.print((int)tdsValue);

  Serial.println(" ppm");                           // نهاية السطر

// =====================================================
// قراءة أوامر التحكم من Firebase
// =====================================================

if (Firebase.getString(fbdo, "/control/mode"))
{
  String mode = fbdo.stringData();

  if (mode == "AUTO") AUTO_MODE = true;
  if (mode == "MANUAL") AUTO_MODE = false;
}

if (!AUTO_MODE)
{
  if (Firebase.getString(fbdo, "/control/fan"))
  {
    digitalWrite(FAN_RELAY, fbdo.stringData() == "ON");
  }

  if (Firebase.getString(fbdo, "/control/pump"))
  {
    digitalWrite(PUMP_RELAY, fbdo.stringData() == "ON");
  }

  if (Firebase.getString(fbdo, "/control/window"))
  {
    digitalWrite(WINDOW_RELAY, fbdo.stringData() == "ON");
  }

  if (Firebase.getString(fbdo, "/control/buzzer"))
  {
    digitalWrite(BUZZER_PIN, fbdo.stringData() == "ON");
  }
}

// =====================================================
// إرسال البيانات إلى Firebase (Dashboard)
// =====================================================

Firebase.setFloat(fbdo, "/sensor/temp", temp);
Firebase.setFloat(fbdo, "/sensor/hum", hum);
Firebase.setFloat(fbdo, "/sensor/ph", phValue);
Firebase.setInt(fbdo, "/sensor/tds", (int)tdsValue);

  // =====================================================  
  // تأخير الدورة
  // =====================================================  

  delay(2000);                                      // انتظار قبل التكرار
}
