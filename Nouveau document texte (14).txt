// =====================================================    // خط فاصل للتنظيم فقط
// البيت الزراعي الذكي - ESP32                            // عنوان المشروع
// DHT22 + DS18B20 + LCD I2C + 3 LEDs (بدل الريليهات)     // المكونات المستخدمة في المشروع
// =====================================================    // نهاية عنوان المشروع

// استدعاء مكتبة الاتصال I2C الخاصة بالشاشة               // تعليق توضيحي
#include <Wire.h>                                           // استدعاء مكتبة Wire المسؤولة عن بروتوكول I2C

// استدعاء مكتبة شاشة LCD I2C                              // تعليق توضيحي
#include <LiquidCrystal_I2C.h>                              // استدعاء مكتبة التحكم في شاشة LCD المتصلة عبر I2C

// استدعاء مكتبة حساس DHT22                                // تعليق توضيحي
#include <DHT.h>                                            // استدعاء مكتبة التعامل مع حساس DHT22

// استدعاء مكتبات حساس DS18B20                             // تعليق توضيحي
#include <OneWire.h>                                        // استدعاء مكتبة بروتوكول OneWire
#include <DallasTemperature.h>                              // استدعاء مكتبة قراءة درجة الحرارة من DS18B20

// =====================================================    // خط فاصل للتنظيم
// إعداد DHT22                                              // بداية إعدادات حساس DHT22
// =====================================================    // خط فاصل للتنظيم

#define DHTPIN 4                 // تعريف أن رجل البيانات الخاصة بحساس DHT22 متصلة بالمنفذ GPIO4
#define DHTTYPE DHT22            // تعريف نوع الحساس المستخدم وهو DHT22

// =====================================================    // خط فاصل للتنظيم
// إعداد أرجل I2C الخاصة بالشاشة                           // بداية إعدادات شاشة LCD
// =====================================================    // خط فاصل للتنظيم

#define SDA_PIN 8                // تعريف الرجل SDA الخاصة بخط البيانات في I2C على GPIO8
#define SCL_PIN 9                // تعريف الرجل SCL الخاصة بخط الساعة في I2C على GPIO9

// =====================================================    // خط فاصل للتنظيم
// إعداد DS18B20                                            // بداية إعدادات حساس الحرارة DS18B20
// =====================================================    // خط فاصل للتنظيم

#define ONE_WIRE_BUS 5           // تعريف أن رجل البيانات الخاصة بحساس DS18B20 متصلة بالمنفذ GPIO5

// =====================================================    // خط فاصل للتنظيم
// إعداد مخارج الـ LEDات (بدل الريليهات مؤقتاً)           // تعريف المخارج المستخدمة لاختبار الـLEDات
// =====================================================    // خط فاصل للتنظيم

#define FAN_RELAY     16         // تعريف منفذ GPIO16 ليستخدم مع LED المروحة
#define PUMP_RELAY    17         // تعريف منفذ GPIO17 ليستخدم مع LED المضخة
#define WINDOW_RELAY  18         // تعريف منفذ GPIO18 ليستخدم مع LED النافذة

// =====================================================    // خط فاصل للتنظيم
// إعداد حساسات pH و TDS                                   // بداية تعريف حساسات المياه
// =====================================================    // خط فاصل للتنظيم

#define PH_PIN   1               // تعريف منفذ GPIO1 كمدخل تماثلي لحساس pH
#define TDS_PIN  2               // تعريف منفذ GPIO2 كمدخل تماثلي لحساس TDS

// =====================================================    // خط فاصل للتنظيم
// إنشاء الكائنات                                           // إنشاء الكائنات الخاصة بالمكتبات
// =====================================================    // خط فاصل للتنظيم

DHT dht(DHTPIN, DHTTYPE);              // إنشاء كائن باسم dht وربطه بالرجل DHTPIN وتحديد نوع الحساس DHT22

OneWire oneWire(ONE_WIRE_BUS);         // إنشاء ناقل OneWire وربطه بالرجل المحددة للحساس

DallasTemperature ds18b20(&oneWire);   // إنشاء كائن ds18b20 وربطه بناقل OneWire السابق

LiquidCrystal_I2C lcd(0x27, 16, 2);    // إنشاء كائن الشاشة بعنوان I2C يساوي 0x27 وبحجم 16 عمود و2 أسطر

// =====================================================    // خط فاصل للتنظيم
// متغيرات عامة                                             // بداية تعريف المتغيرات المستخدمة في البرنامج
// =====================================================    // خط فاصل للتنظيم

float temp = 0;                 // إنشاء متغير عشري لتخزين درجة الحرارة وتهيئته بالصفر

float hum = 0;                  // إنشاء متغير عشري لتخزين نسبة الرطوبة وتهيئته بالصفر

bool windowClosed = false;      // إنشاء متغير منطقي لمعرفة هل النافذة مغلقة أم لا، والقيمة الابتدائية هي لا

int phRaw = 0;                  // إنشاء متغير صحيح لتخزين القراءة الخام لحساس pH

int tdsRaw = 0;                 // إنشاء متغير صحيح لتخزين القراءة الخام لحساس TDS

unsigned long lastScreenChange = 0;    // إنشاء متغير لتخزين آخر وقت تم فيه تغيير صفحة الشاشة

bool screenPage = false;               // إنشاء متغير منطقي لتحديد الصفحة الحالية المعروضة على الشاشة

// =====================================================    // خط فاصل للتنظيم
// الدالة Setup                                             // بداية دالة الإعداد التي تنفذ مرة واحدة عند تشغيل المتحكم
// =====================================================    // خط فاصل للتنظيم

void setup()                        // تعريف دالة setup
{                                   // بداية الدالة

  Serial.begin(115200);             // تشغيل الاتصال التسلسلي بسرعة 115200 بت في الثانية

  Wire.begin(SDA_PIN, SCL_PIN);     // تهيئة ناقل I2C باستخدام الأرجل المحددة سابقاً

  Wire.setClock(100000);            // ضبط سرعة اتصال I2C على 100 كيلوهرتز

  dht.begin();                      // تهيئة وتشغيل حساس DHT22

  ds18b20.begin();                  // تهيئة وتشغيل حساس DS18B20

  lcd.init();                       // تهيئة شاشة LCD وتجهيزها للعمل

  lcd.backlight();                  // تشغيل الإضاءة الخلفية للشاشة

  pinMode(FAN_RELAY, OUTPUT);       // تعريف منفذ LED المروحة كمخرج رقمي

  pinMode(PUMP_RELAY, OUTPUT);      // تعريف منفذ LED المضخة كمخرج رقمي

  pinMode(WINDOW_RELAY, OUTPUT);    // تعريف منفذ LED النافذة كمخرج رقمي

  digitalWrite(FAN_RELAY, LOW);     // إطفاء LED المروحة عند بدء التشغيل

  digitalWrite(PUMP_RELAY, LOW);    // إطفاء LED المضخة عند بدء التشغيل

  digitalWrite(WINDOW_RELAY, LOW);  // إطفاء LED النافذة عند بدء التشغيل

  lcd.clear();                      // مسح أي بيانات قديمة من الشاشة

  lcd.setCursor(0, 0);              // وضع مؤشر الكتابة في العمود 0 والسطر 0

  lcd.print("SMART GREEN");         // طباعة النص SMART GREEN على السطر الأول

  lcd.setCursor(0, 1);              // نقل المؤشر إلى العمود 0 والسطر 1

  lcd.print("HOUSE");               // طباعة النص HOUSE على السطر الثاني

  delay(2000);                      // الانتظار لمدة ثانيتين حتى يتمكن المستخدم من قراءة رسالة البداية

}                                   // نهاية دالة setup

// =====================================================    // خط فاصل للتنظيم
// الدالة Loop                                              // بداية الدالة الرئيسية التي تتكرر باستمرار
// =====================================================    // خط فاصل للتنظيم

void loop()                                                // تعريف الدالة loop
{                                                          // بداية الدالة

  // =====================================================  // خط فاصل للتنظيم
  // قراءة القيم من الحساسات                               // بداية قسم قراءة الحساسات
  // =====================================================  // خط فاصل للتنظيم

  hum = dht.readHumidity();                                // قراءة قيمة الرطوبة من حساس DHT22 وتخزينها في المتغير hum

  ds18b20.requestTemperatures();                           // إرسال أمر إلى حساس DS18B20 ليقوم بقياس درجة الحرارة

  temp = ds18b20.getTempCByIndex(0);                       // قراءة درجة الحرارة من أول حساس DS18B20 وتخزينها في المتغير temp

  phRaw = analogRead(PH_PIN);                              // قراءة القيمة التماثلية الخام من حساس pH وتخزينها في المتغير phRaw

  tdsRaw = analogRead(TDS_PIN);                            // قراءة القيمة التماثلية الخام من حساس TDS وتخزينها في المتغير tdsRaw

  // =====================================================  // خط فاصل للتنظيم
  // التحقق من سلامة القراءات                              // التأكد من أن الحساسات تعمل بشكل صحيح
  // =====================================================  // خط فاصل للتنظيم

  if (isnan(hum) || temp == DEVICE_DISCONNECTED_C)         // إذا كانت قراءة الرطوبة غير صالحة أو حساس الحرارة غير متصل
  {                                                        // بداية شرط التحقق من الأخطاء

    lcd.setCursor(0, 0);                                   // وضع مؤشر الشاشة في بداية السطر الأول

    lcd.print("SENSOR ERROR    ");                         // عرض رسالة تفيد بوجود خطأ في أحد الحساسات

    lcd.setCursor(0, 1);                                   // نقل المؤشر إلى بداية السطر الثاني

    lcd.print("CHECK SENSORS   ");                         // عرض رسالة تطلب التحقق من الحساسات

    Serial.println("Sensor Error");                        // إرسال رسالة خطأ إلى نافذة Serial Monitor

    delay(2000);                                           // الانتظار لمدة ثانيتين

    return;                                                // إنهاء الدورة الحالية من loop والعودة إلى بدايتها

  }                                                        // نهاية شرط التحقق من الأخطاء

  // =====================================================  // خط فاصل للتنظيم
  // التحكم في المروحة والنافذة                            // بداية قسم التحكم في LED المروحة و LED النافذة
  // =====================================================  // خط فاصل للتنظيم

  if (temp < 25)                                           // إذا كانت درجة الحرارة أقل من 25 درجة مئوية
  {                                                        // بداية الشرط

    digitalWrite(FAN_RELAY, LOW);                          // إطفاء LED المروحة لأن الحرارة ضمن المجال الطبيعي

    if (!windowClosed)                                     // إذا كانت النافذة لم تغلق سابقاً
    {                                                      // بداية الشرط الداخلي

      digitalWrite(WINDOW_RELAY, HIGH);                    // تشغيل LED النافذة لمحاكاة عمل محرك النافذة

      delay(3000);                                         // إبقاء LED مضاءً لمدة 3 ثوان

      digitalWrite(WINDOW_RELAY, LOW);                     // إطفاء LED النافذة بعد انتهاء الحركة

      windowClosed = true;                                 // تسجيل أن النافذة أصبحت مغلقة

    }                                                      // نهاية الشرط الداخلي

  }                                                        // نهاية شرط temp < 25

  else                                                     // إذا كانت درجة الحرارة أكبر أو تساوي 25 درجة
  {                                                        // بداية فرع else

    digitalWrite(FAN_RELAY, HIGH);                         // تشغيل LED المروحة للدلالة على تشغيل المروحة

    windowClosed = false;                                  // إعادة المتغير إلى false للسماح بتنفيذ حركة النافذة مستقبلاً

  }                                                        // نهاية فرع else

  // =====================================================  // خط فاصل للتنظيم
  // التحكم في المضخة حسب الرطوبة                          // بداية قسم التحكم في LED المضخة
  // =====================================================  // خط فاصل للتنظيم

  if (hum < 40)                                            // إذا كانت نسبة الرطوبة أقل من 40%
  {                                                        // بداية الشرط

    digitalWrite(PUMP_RELAY, HIGH);                        // تشغيل LED المضخة للدلالة على تشغيل المضخة

  }                                                        // نهاية شرط الرطوبة المنخفضة

  else                                                     // إذا كانت الرطوبة 40% أو أكثر
  {                                                        // بداية فرع else

    digitalWrite(PUMP_RELAY, LOW);                         // إطفاء LED المضخة لأن الري غير مطلوب

  }                                                        // نهاية فرع else

    // =====================================================  // خط فاصل للتنظيم
  // عرض البيانات على الشاشة (صفحتان)                     // بداية قسم التحكم في صفحات شاشة LCD
  // =====================================================  // خط فاصل للتنظيم

  // تبديل الصفحة كل 3 ثوان                               // تعليق يوضح وظيفة الشرط التالي
  if (millis() - lastScreenChange >= 3000UL)              // إذا مرّت 3000 ميلي ثانية (3 ثوان) منذ آخر تغيير للصفحة
  {                                                       // بداية الشرط

    lastScreenChange = millis();                          // حفظ الوقت الحالي ليصبح هو وقت آخر تغيير للصفحة

    // تبديل الصفحة بين الأولى والثانية                   // تعليق توضيحي
    if (screenPage == false)                              // إذا كانت الصفحة الحالية هي الصفحة الأولى
      screenPage = true;                                  // الانتقال إلى الصفحة الثانية
    else                                                  // إذا لم تكن الصفحة الأولى (أي أنها الصفحة الثانية)
      screenPage = false;                                 // العودة إلى الصفحة الأولى

    lcd.clear();                                          // مسح الشاشة مرة واحدة عند تغيير الصفحة لتجنب بقاء بيانات قديمة

  }                                                       // نهاية شرط تغيير الصفحة

  // =====================================================  // خط فاصل للتنظيم
  // الصفحة الأولى : عرض الحرارة والرطوبة                 // بداية كود الصفحة الأولى
  // =====================================================  // خط فاصل للتنظيم

  if (screenPage == false)                                // إذا كانت الصفحة الحالية هي الصفحة الأولى
  {                                                       // بداية الشرط

    lcd.setCursor(0, 0);                                  // وضع مؤشر الكتابة في العمود 0 والسطر 0

    lcd.print("T:");                                      // طباعة الحرف T متبوعًا بنقطتين للدلالة على درجة الحرارة

    lcd.print(temp, 1);                                   // طباعة قيمة درجة الحرارة مع إظهار منزلة عشرية واحدة

    lcd.print("C ");                                      // طباعة الرمز C للدلالة على وحدة الدرجة المئوية

    lcd.setCursor(9, 0);                                  // نقل المؤشر إلى العمود 9 من السطر الأول

    lcd.print("H:");                                      // طباعة الحرف H متبوعًا بنقطتين للدلالة على الرطوبة

    lcd.print(hum, 0);                                    // طباعة قيمة الرطوبة بدون أي منازل عشرية

    lcd.print("% ");                                      // طباعة علامة النسبة المئوية بعد قيمة الرطوبة

    lcd.setCursor(0, 1);                                  // نقل المؤشر إلى بداية السطر الثاني

    // =================================================  // خط فاصل للتوضيح
    // عرض حالة درجة الحرارة                             // تحديد وصف للحالة حسب درجة الحرارة
    // =================================================  // خط فاصل للتوضيح

    if (temp >= 40)                                       // إذا كانت درجة الحرارة أكبر من أو تساوي 40 درجة
    {                                                     // بداية الشرط

      lcd.print("HIGH TEMP !    ");                       // عرض رسالة تفيد بأن درجة الحرارة مرتفعة جدًا

    }                                                     // نهاية الشرط الأول

    else if (temp >= 30)                                  // إذا كانت الحرارة أقل من 40 وأكبر من أو تساوي 30
    {                                                     // بداية الشرط الثاني

      lcd.print("MEDIUM TEMP    ");                       // عرض رسالة تفيد بأن الحرارة متوسطة

    }                                                     // نهاية الشرط الثاني

    else if (temp >= 25)                                  // إذا كانت الحرارة أقل من 30 وأكبر من أو تساوي 25
    {                                                     // بداية الشرط الثالث

      lcd.print("LOW TEMP       ");                       // عرض رسالة تفيد بأن الحرارة منخفضة ولكن فوق الطبيعي المحدد

    }                                                     // نهاية الشرط الثالث

    else                                                  // إذا كانت الحرارة أقل من 25 درجة
    {                                                     // بداية فرع else

      lcd.print("NORMAL         ");                       // عرض رسالة تفيد بأن درجة الحرارة طبيعية

    }                                                     // نهاية فرع else

  }                                                       // نهاية شرط الصفحة الأولى

  // =====================================================  // خط فاصل للتنظيم
  // الصفحة الثانية : عرض pH و TDS                        // بداية كود الصفحة الثانية
  // =====================================================  // خط فاصل للتنظيم

  else                                                    // إذا كانت الصفحة الحالية هي الصفحة الثانية
  {                                                       // بداية فرع else

    lcd.setCursor(0, 0);                                  // وضع مؤشر الكتابة في بداية السطر الأول

    lcd.print("pH RAW:");                                 // طباعة النص pH RAW على الشاشة

    lcd.print(phRaw);                                     // طباعة القيمة الخام المقروءة من حساس pH

    lcd.print("    ");                                    // طباعة مسافات إضافية لمسح أي أرقام قديمة متبقية

    lcd.setCursor(0, 1);                                  // نقل المؤشر إلى بداية السطر الثاني

    lcd.print("TDS:");                                    // طباعة النص TDS على الشاشة

    lcd.print(tdsRaw);                                    // طباعة القيمة الخام المقروءة من حساس TDS

    lcd.print("       ");                                 // طباعة مسافات إضافية لمسح أي أرقام قديمة متبقية

  }                                                       // نهاية فرع الصفحة الثانية

    // =====================================================  // خط فاصل للتنظيم
  // إرسال البيانات إلى Serial Monitor                    // بداية قسم إرسال القراءات إلى المنفذ التسلسلي
  // =====================================================  // خط فاصل للتنظيم

  Serial.print("Temperature = ");                          // إرسال النص "Temperature = " إلى نافذة Serial Monitor بدون الانتقال إلى سطر جديد

  Serial.print(temp);                                      // إرسال قيمة درجة الحرارة المخزنة في المتغير temp

  Serial.print(" C  Humidity = ");                         // إرسال النص " C  Humidity = " بعد قيمة الحرارة لإظهار وحدة القياس ثم عنوان الرطوبة

  Serial.print(hum);                                       // إرسال قيمة الرطوبة المخزنة في المتغير hum

  Serial.print(" %  pH_RAW = ");                           // إرسال النص " %  pH_RAW = " بعد قيمة الرطوبة لإظهار علامة النسبة المئوية ثم عنوان قراءة pH

  Serial.print(phRaw);                                     // إرسال القيمة الخام المقروءة من حساس pH

  Serial.print("  TDS_RAW = ");                            // إرسال النص "  TDS_RAW = " بعد قيمة pH لإظهار عنوان قراءة TDS

  Serial.println(tdsRaw);                                  // إرسال القيمة الخام المقروءة من حساس TDS ثم الانتقال إلى سطر جديد في نافذة Serial Monitor

  // =====================================================  // خط فاصل للتنظيم
  // تأخير قبل إعادة الدورة                               // بداية قسم الانتظار قبل تكرار البرنامج
  // =====================================================  // خط فاصل للتنظيم

  delay(2000);                                             // إيقاف تنفيذ البرنامج لمدة 2000 ميلي ثانية (ثانيتين) قبل إعادة تنفيذ الدالة loop من البداية

}
