void readSensors() {

 float newH = dht.readHumidity();
float newT = dht.readTemperature();

if (!isnan(newH))
  h = newH;

if (!isnan(newT))
  t = newT;

  ds18b20.requestTemperatures();
  float temp = ds18b20.getTempCByIndex(0);

if (temp != DEVICE_DISCONNECTED_C)
{
  dsTemp = temp;
}

  ph = 0;
tds = 0;
}
